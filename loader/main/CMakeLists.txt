# The partition tables the loader may write, generated from the same CSVs the
# project builds with so there is one source of truth. They differ only in
# storage's size, which is the point: the table is a size fix-up for the flash
# the board actually has.
#
# The set of sizes is per target. Adding a target means adding its sizes here
# and its partitions/<target>-*.csv files; nothing else in the loader changes.
set(espix_gen "${IDF_PATH}/components/partition_table/gen_esp32part.py")
set(espix_out "${CMAKE_CURRENT_BINARY_DIR}")

if(CONFIG_IDF_TARGET STREQUAL "esp32s3")
    set(espix_pt_sizes 8mb 16mb)
elseif(CONFIG_IDF_TARGET STREQUAL "esp32s31")
    set(espix_pt_sizes 8mb 16mb 32mb)
else()
    # IDF evaluates a component's CMakeLists twice: an early pass that only
    # resolves REQUIRES, before CONFIG_ values exist, and the real pass. Empty
    # here means the early pass -- there is nothing to generate yet, and failing
    # would fail every build.
    set(espix_pt_sizes)
endif()

if(espix_pt_sizes)
    set(espix_pt_emit)
    set(espix_pt_entries "")
    foreach(size ${espix_pt_sizes})
        execute_process(
            COMMAND "${PYTHON}" "${espix_gen}" -q
                    "${CMAKE_CURRENT_LIST_DIR}/../../partitions/${CONFIG_IDF_TARGET}-${size}.csv"
                    "${espix_out}/pt-${size}.bin"
            RESULT_VARIABLE rc)
        if(NOT rc EQUAL 0)
            message(FATAL_ERROR "loader: cannot generate the ${CONFIG_IDF_TARGET}-${size} table")
        endif()
        # "8mb" -> 8, for the byte size the table covers.
        string(REGEX REPLACE "mb$" "" mb "${size}")
        list(APPEND espix_pt_emit espix_pt_${size} "${espix_out}/pt-${size}.bin")
        string(APPEND espix_pt_entries
            "    { ${mb}u * 1024u * 1024u, espix_pt_${size}, espix_pt_${size}_len },\n")
    endforeach()

    execute_process(
        COMMAND "${PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/../../tools/emit-array.py"
                "${espix_out}/espix_partition_tables.h"
                ${espix_pt_emit}
        RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0)
        message(FATAL_ERROR "loader: cannot emit the partition tables")
    endif()

    # An index over the arrays above, ascending by size, so loader_main.c picks a
    # table by asking "the largest one that fits this flash" rather than naming a
    # table per target.
    file(WRITE "${espix_out}/espix_pt_index.h"
    "/* Generated by loader/main/CMakeLists.txt -- do not edit. */\n"
    "#pragma once\n"
    "#include \"espix_partition_tables.h\"\n"
    "typedef struct {\n"
    "    unsigned bytes;\n"
    "    const unsigned char *data;\n"
    "    unsigned len;\n"
    "} espix_pt_entry_t;\n"
    "/* Ascending by size. */\n"
    "static const espix_pt_entry_t espix_pt_tables[] = {\n"
    ${espix_pt_entries}
    "};\n"
    "#define ESPIX_PT_TABLE_COUNT (sizeof(espix_pt_tables) / sizeof(espix_pt_tables[0]))\n")
endif()

idf_component_register(
    SRCS
        "loader_main.c"
    INCLUDE_DIRS
        "."
        "${CMAKE_CURRENT_BINARY_DIR}"
    REQUIRES
        esp_partition
        app_update
        esp_app_format
        bootloader_support
        spi_flash
        freertos
        nvs_flash
        joltwallet__littlefs
)
